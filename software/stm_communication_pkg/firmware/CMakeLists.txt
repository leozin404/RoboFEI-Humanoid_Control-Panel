# =============================================================================
# RoboFEI Control Panel — STM32U575CGT6 firmware build
#
# This CMakeLists.txt is intentionally OUTSIDE the ROS2/ament build graph.
# The COLCON_IGNORE file in this same directory prevents `colcon build`
# from ever descending into this folder — it is built separately, with the
# arm-none-eabi toolchain, typically from STM32CubeIDE or invoked manually
# as shown below.
# =============================================================================

cmake_minimum_required(VERSION 3.22)

# ---------------------------------------------------------------------------
# Toolchain — must be set BEFORE project(). If you're using STM32CubeIDE,
# it generates its own build system and this file may only be needed for
# CLI builds / CI. Adjust the path below to your arm-none-eabi-gcc install.
# ---------------------------------------------------------------------------
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)

set(CMAKE_C_COMPILER   arm-none-eabi-gcc)
set(CMAKE_CXX_COMPILER arm-none-eabi-g++)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_OBJCOPY      arm-none-eabi-objcopy)
set(CMAKE_SIZE         arm-none-eabi-size)

set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

project(control_panel_firmware C CXX ASM)

# ---------------------------------------------------------------------------
# C++ standard — REQUIRED: button_fsm.hpp uses std::optional (C++17).
# Do not lower this; the FSM's update() return type depends on it.
# ---------------------------------------------------------------------------
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# ---------------------------------------------------------------------------
# MCU-specific flags — STM32U575CGT6 (Cortex-M33, single precision FPU)
# ---------------------------------------------------------------------------
set(MCU_FLAGS
    -mcpu=cortex-m33
    -mthumb
    -mfpu=fpv5-sp-d16
    -mfloat-abi=hard
)

add_compile_options(
    ${MCU_FLAGS}
    -Wall
    -Wextra
    -ffunction-sections
    -fdata-sections
    -fno-exceptions      # bare-metal: no exception unwinding support
    -fno-rtti            # not needed, saves flash
    $<$<CONFIG:Debug>:-Og -g3>
    $<$<CONFIG:Release>:-Os>
)

add_link_options(
    ${MCU_FLAGS}
    -Wl,--gc-sections
    -specs=nano.specs
    -specs=nosys.specs
    -T${CMAKE_CURRENT_SOURCE_DIR}/STM32U575CGTX_FLASH.ld   # adjust to your linker script
)

add_compile_definitions(
    STM32U575xx
    USE_HAL_DRIVER
)

# ---------------------------------------------------------------------------
# Include directories
# ---------------------------------------------------------------------------
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}/Core/Inc
    # Add HAL / CMSIS / USB middleware include paths here once generated
    # by CubeMX, e.g.:
    # ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/STM32U5xx_HAL_Driver/Inc
    # ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS/Device/ST/STM32U5xx/Include
    # ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS/Include
    # ${CMAKE_CURRENT_SOURCE_DIR}/USB_DEVICE/App
    # ${CMAKE_CURRENT_SOURCE_DIR}/USB_DEVICE/Target
    # ${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/ST/STM32_USB_Device_Library/Core/Inc
    # ${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/ST/STM32_USB_Device_Library/Class/CDC/Inc
)

# ---------------------------------------------------------------------------
# Sources
# ---------------------------------------------------------------------------
set(APP_SOURCES
    Core/Src/main.cpp
    Core/Src/button_fsm.cpp
    Core/Src/led_driver.cpp
    # Core/Src/stm32u5xx_hal_msp.c        # add once generated by CubeMX
    # Core/Src/stm32u5xx_it.c             # interrupt handlers
    # Core/Src/system_stm32u5xx.c         # CMSIS system init
    # USB_DEVICE/App/usbd_cdc_if.c        # CDC receive/transmit callbacks
    # USB_DEVICE/App/usb_device.c
)

# HAL driver sources (uncomment/expand once CubeMX generates them):
# file(GLOB HAL_SOURCES
#     "${CMAKE_CURRENT_SOURCE_DIR}/Drivers/STM32U5xx_HAL_Driver/Src/*.c"
# )
# list(APPEND APP_SOURCES ${HAL_SOURCES})

set(STARTUP_SOURCE
    ${CMAKE_CURRENT_SOURCE_DIR}/startup_stm32u575xx.s   # from CubeMX / CMSIS
)

add_executable(${PROJECT_NAME}.elf
    ${APP_SOURCES}
    ${STARTUP_SOURCE}
)

# ---------------------------------------------------------------------------
# Post-build: generate .bin/.hex and print size report
# ---------------------------------------------------------------------------
add_custom_command(TARGET ${PROJECT_NAME}.elf POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O ihex   $<TARGET_FILE:${PROJECT_NAME}.elf> ${PROJECT_NAME}.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${PROJECT_NAME}.elf> ${PROJECT_NAME}.bin
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${PROJECT_NAME}.elf>
    COMMENT "Generating .hex/.bin and size report"
)
